Step 1: Check Kubernetes version
kubectl version --client


Step 2: Check Kubernetes nodes
kubectl get nodes


Step 3: Create deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mywebsite-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: mywebsite
  template:
    metadata:
      labels:
        app: mywebsite
    spec:
      containers:
      - name: mywebsite
        image: sagar05123/mywebsite:v1
        ports:
        - containerPort: 80


Step 4: Apply deployment
kubectl apply -f deployment.yaml


Step 5: Create service.yaml
apiVersion: v1
kind: Service
metadata:
  name: mywebsite-service
spec:
  type: NodePort
  selector:
    app: mywebsite
  ports:
  - port: 80
    targetPort: 80
    nodePort: 30080


Step 6: Apply service
kubectl apply -f service.yaml


Step 7: Check Pods
kubectl get pods


Step 8: Scale deployment to 4
kubectl scale deployment mywebsite-deployment --replicas=4


Step 9: Check Pods again
kubectl get pods


Step 10: Delete one Pod
kubectl delete pod POD_NAME


Replace POD_NAME with an actual pod name.
Step 11: Check Pods again
kubectl get pods


Kubernetes automatically creates a replacement Pod.
Step 12: Check endpoints
kubectl get endpoints mywebsite-service


Step 13: Check services
kubectl get services


Step 14: Check all resources
kubectl get all
